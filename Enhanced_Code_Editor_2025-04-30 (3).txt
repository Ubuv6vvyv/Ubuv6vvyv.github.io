class JSONQuery {
    // JSONQuery class definition remains the same as the previous version
    // including: constructor, findAllRecursive, searchForKey, findKeysInItem,
    // parse, parseCondition, parseProperty, parseMapObject, parseObject,
    // splitCommas, execute, evaluateCondition, getValue
    constructor() {
        this.operators = { /* ... operators ... */ };
        this.operators = { '==': 'eq', '!=': 'neq', '>': 'gt', '>=': 'gte', '<': 'lt', '<=': 'lte', 'and': 'and', 'or': 'or', 'in': 'in', 'not in': 'notIn' };
    }
    findAllRecursive(data, keyToFind) { /* ... */
        let results = []; if (Array.isArray(data)) { data.forEach(item => { results = results.concat(this.findAllRecursive(item, keyToFind)); }); } else if (typeof data === 'object' && data !== null) { for (const key in data) { if (data.hasOwnProperty(key)) { if (key === keyToFind) { results.push(data[key]); } results = results.concat(this.findAllRecursive(data[key], keyToFind)); } } } return results;
    }
    searchForKey(data, keyName) { /* ... */
        if (typeof data !== 'object' || data === null) { return undefined; } if (!Array.isArray(data)) { if (data.hasOwnProperty(keyName)) { return data[keyName]; } } for (const key in data) { if (data.hasOwnProperty(key)) { const found = this.searchForKey(data[key], keyName); if (found !== undefined) { return found; } } } return undefined;
    }
    findKeysInItem(item, keysToFind) { /* ... */
        const result = {}; let foundAny = false; for (const keyOrPath of keysToFind) { if (typeof keyOrPath === 'string') { const value = this.searchForKey(item, keyOrPath); if (value !== undefined) { result[keyOrPath] = value; foundAny = true; } } else if (Array.isArray(keyOrPath) && keyOrPath[0] === 'get') { const pathKey = keyOrPath[keyOrPath.length - 1]; const value = this.getValue(item, keyOrPath); if (value !== undefined) { result[pathKey] = value; foundAny = true; } } } return foundAny ? result : null;
    }
    parse(q) { /* ... same robust parse ... */
        try { q = q.trim(); if (q.includes('|')) { const p = q.split('|').map(t => t.trim()).filter(t => t !== ''); return ['pipe', ...p.map(t => this.parse(t))]; } const funcMatch = q.match(/^([a-zA-Z_][a-zA-Z0-9_]*)\((.*)\)$/); if (funcMatch) { const funcName = funcMatch[1]; const argsString = funcMatch[2].trim(); let args = []; if (argsString) { if (funcName === 'extract') { args = this.splitCommas(argsString).map(arg => { arg = arg.trim(); if (arg.startsWith('.')) { return this.parseProperty(arg); } else if ((arg.startsWith("'") && arg.endsWith("'")) || (arg.startsWith('"') && arg.endsWith('"'))) { return arg.substring(1, arg.length - 1); } else { throw new Error(`Invalid argument in extract: ${arg}. Use quotes for key names ('key') or dot for paths (.path.key)`); } }); } else if (funcName === 'findAll') { if ((argsString.startsWith("'") && argsString.endsWith("'")) || (argsString.startsWith('"') && argsString.endsWith('"'))) { args = [argsString.substring(1, argsString.length - 1)]; } else { throw new Error(`Invalid key argument for findAll: ${argsString}. Key name must be enclosed in single or double quotes.`); } } else if (funcName === 'filter') { args = [this.parseCondition(argsString)]; } else if (funcName === 'sort') { const sortArgs = this.splitCommas(argsString).map(t => t.trim()); args.push(this.parseProperty(sortArgs[0])); if (sortArgs.length > 1) { args.push(sortArgs[1].replace(/['"]/g, '')); } } else if (funcName === 'group') { args = [this.parseProperty(argsString)]; } else if (funcName === 'pick') { args = this.splitCommas(argsString).map(t => this.parseProperty(t.trim())); } } if (funcName === 'map') { if (argsString.startsWith('{') && argsString.endsWith('}')) { return ['map', ['object', this.parseMapObject(argsString)]]; } else if (argsString.startsWith('.')) { return ['map', this.parseProperty(argsString)]; } else { throw new Error(`Invalid map argument: ${argsString}`); } } else if (funcName === 'size' || funcName === 'average') { return [funcName]; } else if (args.length > 0 || ['findAll', 'extract', 'filter', 'sort', 'group', 'pick'].includes(funcName)) { return [funcName, ...args]; } else { throw new Error(`Unrecognized function or arguments: ${q}`); } } if (q.startsWith('.')) { return this.parseProperty(q); } if (q.startsWith('{') && q.endsWith('}')) { return ['object', this.parseObject(q)]; } throw new Error(`Unrecognized query segment: ${q}`); } catch (e) { const message = e.message || String(e); throw new Error(`${message} (in query segment: "${q}")`); }
    }
    parseCondition(c) { /* ... */
        for (const opStr in this.operators) { const opRegex = new RegExp(`\\s${opStr.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')}\\s`); if (opRegex.test(` ${c} `)) { let parts = c.split(opRegex); if (parts.length === 2) { let [l, r] = parts.map(t => t.trim()); const parseSide = (side) => { if (side.startsWith('.')) { return this.parseProperty(side); } if (side === 'true' || side === 'false') { return side === 'true'; } if (side === 'null') { return null; } if (!isNaN(Number(side)) && side.trim() !== '') { return Number(side); } if ((side.startsWith("'") && side.endsWith("'")) || (side.startsWith('"') && side.endsWith('"'))) { return side.substring(1, side.length - 1); } throw new Error(`Invalid value in condition: ${side}. Use quotes for strings.`); }; l = parseSide(l); r = parseSide(r); return [this.operators[opStr], l, r]; } } } throw new Error(`Invalid condition structure or unsupported operator: ${c}`);
    }
    parseProperty(p) { /* ... */
        if (!p.startsWith('.')) { throw new Error(`Invalid property access: ${p}. Must start with '.'`); } const tokens = p.substring(1).split('.'); return ['get', ...tokens.filter(t => t)];
    }
    parseMapObject(o) { /* ... */
        try { if (!o.startsWith('{') || !o.endsWith('}')) throw new Error("Map object must be enclosed in {}"); const content = o.substring(1, o.length - 1).trim(); if (content === '') return {}; const result = {}; const pairs = this.splitCommas(content); for (const pair of pairs) { const colonIndex = pair.indexOf(':'); if (colonIndex === -1) throw new Error(`Invalid key-value pair in map object: ${pair}`); let key = pair.substring(0, colonIndex).trim(); const valueString = pair.substring(colonIndex + 1).trim(); if ((key.startsWith("'") && key.endsWith("'")) || (key.startsWith('"') && key.endsWith('"'))) { key = key.substring(1, key.length - 1); } else if (!/^[a-zA-Z_][a-zA-Z0-9_]*$/.test(key)) { throw new Error(`Invalid key format in map object: ${key}. Use quotes if needed.`); } result[key] = this.parse(valueString); } return result; } catch (e) { throw new Error(`Map object parse error: ${e.message || e} in "${o}"`); }
    }
    parseObject(o) { /* ... */
        try { const content = o.substring(1, o.length - 1).trim(); if (content === '') return {}; const result = {}; const pairs = this.splitCommas(content); for (const pair of pairs) { const colonIndex = pair.indexOf(':'); if (colonIndex === -1) throw new Error(`Invalid key-value pair in object literal: ${pair}`); let key = pair.substring(0, colonIndex).trim(); const valueString = pair.substring(colonIndex + 1).trim(); if ((key.startsWith("'") && key.endsWith("'")) || (key.startsWith('"') && key.endsWith('"'))) { key = key.substring(1, key.length - 1); } result[key] = this.parse(valueString); } return result; } catch (e) { throw new Error(`Object literal parse error: ${e.message} in "${o}"`); }
    }
    splitCommas(s) { /* ... */
        const r = []; let current = ''; let braceDepth = 0; let bracketDepth = 0; let parenDepth = 0; let inSingleQuotes = false; let inDoubleQuotes = false; for (let i = 0; i < s.length; i++) { const char = s[i]; current += char; if (char === "'" && s[i - 1] !== '\\') inSingleQuotes = !inSingleQuotes; if (char === '"' && s[i - 1] !== '\\') inDoubleQuotes = !inDoubleQuotes; if (!inSingleQuotes && !inDoubleQuotes) { if (char === '{') braceDepth++; else if (char === '}') braceDepth--; else if (char === '[') bracketDepth++; else if (char === ']') bracketDepth--; else if (char === '(') parenDepth++; else if (char === ')') parenDepth--; else if (char === ',' && braceDepth === 0 && bracketDepth === 0 && parenDepth === 0) { r.push(current.substring(0, current.length - 1).trim()); current = ''; } } } if (current.trim()) { r.push(current.trim()); } return r;
    }
    execute(d, q) { /* ... same robust execute including 'extract' case ... */
         try { const p = typeof q === 'string' ? this.parse(q) : q; if (Array.isArray(p)) { const [op, ...args] = p; switch (op) { case 'pipe': return args.reduce((currentData, queryPart) => this.execute(currentData, queryPart), d); case 'extract': { const keysToFind = args; if (!Array.isArray(keysToFind) || keysToFind.length === 0) { throw new Error("Extract requires at least one key name or path argument."); } if (Array.isArray(d)) { return d.map(item => this.findKeysInItem(item, keysToFind)).filter(result => result !== null); } else if (typeof d === 'object' && d !== null) { const result = this.findKeysInItem(d, keysToFind); return result === null ? {} : result; } else { throw new Error("Extract can only be applied to objects or arrays of objects."); } } case 'filter': if (!Array.isArray(d)) throw new Error("Filter can only be applied to arrays."); return d.filter(item => this.evaluateCondition(item, args[0])); case 'sort': { if (!Array.isArray(d)) throw new Error("Sort can only be applied to arrays."); const [propGetter, direction = 'asc'] = args; if (typeof propGetter !== 'object' || propGetter[0] !== 'get') { throw new Error("Sort requires a property path (e.g., .name)"); } return [...d].sort((a, b) => { const valA = this.getValue(a, propGetter); const valB = this.getValue(b, propGetter); const comparison = (valA < valB) ? -1 : ((valA > valB) ? 1 : 0); if (valA === undefined || valA === null) return 1; if (valB === undefined || valB === null) return -1; return direction.toLowerCase() === 'desc' ? (comparison * -1) : comparison; }); } case 'map': { if (!Array.isArray(d)) throw new Error("Map can only be applied to arrays."); const mapper = args[0]; if (mapper[0] === 'object') { const objTemplate = mapper[1]; return d.map(item => { const newItem = {}; for (const key in objTemplate) { newItem[key] = this.execute(item, objTemplate[key]); } return newItem; }); } else { return d.map(item => this.getValue(item, mapper)); } } case 'findAll': { const keyName = args[0]; if (typeof keyName !== 'string') throw new Error("findAll requires a key name string."); return this.findAllRecursive(d, keyName); } case 'pick': { if (!Array.isArray(d)) throw new Error("Pick can only be applied to arrays."); return d.map(item => { const newItem = {}; args.forEach(propGetter => { if (typeof propGetter === 'object' && propGetter[0] === 'get' && propGetter.length > 1) { const key = propGetter[propGetter.length - 1]; newItem[key] = this.getValue(item, propGetter); } }); return newItem; }); } case 'group': { if (!Array.isArray(d)) throw new Error("Group can only be applied to arrays."); const propGetter = args[0]; if (typeof propGetter !== 'object' || propGetter[0] !== 'get') { throw new Error("Group requires a property path (e.g., .category)"); } const groups = {}; d.forEach(item => { const key = this.getValue(item, propGetter); const keyStr = typeof key === 'object' ? JSON.stringify(key) : String(key); if (!groups[keyStr]) { groups[keyStr] = { key: key, group: [] }; } groups[keyStr].group.push(item); }); return Object.values(groups); } case 'size': return Array.isArray(d) ? d.length : (typeof d === 'object' && d !== null ? Object.keys(d).length : 0); case 'average': if (!Array.isArray(d)) throw new Error("Average requires an array of numbers."); const numbers = d.filter(n => typeof n === 'number'); return numbers.length === 0 ? 0 : numbers.reduce((sum, val) => sum + val, 0) / numbers.length; case 'object': const newObj = {}; const template = args[0]; for (const key in template) { newObj[key] = this.execute(d, template[key]); } return newObj; case 'get': return this.getValue(d, p); default: throw new Error(`Unknown operation: ${op}`); } } else { return p; } } catch (e) { throw new Error(`Execution error: ${e.message || e}`); }
    }
    evaluateCondition(item, condition) { /* ... */
        if (!Array.isArray(condition)) return !!condition; const [op, left, right] = condition; const valLeft = this.getValue(item, left); const valRight = this.getValue(item, right); switch (op) { case 'eq': return valLeft === valRight; case 'neq': return valLeft !== valRight; case 'gt': return valLeft > valRight; case 'gte': return valLeft >= valRight; case 'lt': return valLeft < valRight; case 'lte': return valLeft <= valRight; case 'and': return this.evaluateCondition(item, left) && this.evaluateCondition(item, right); case 'or': return this.evaluateCondition(item, left) || this.evaluateCondition(item, right); case 'in': return Array.isArray(valRight) && valRight.some(el => el === valLeft); case 'notIn': return Array.isArray(valRight) && !valRight.some(el => el === valLeft); default: return false; }
    }
    getValue(data, getter) { /* ... */
        if (!Array.isArray(getter)) { return getter; } const [op, ...args] = getter; if (op === 'get') { let current = data; for (const prop of args) { if (current === null || typeof current !== 'object') return undefined; current = current[prop]; } return current; } else { return this.execute(data, getter); }
    }
} // End Class

function findAllUniqueKeysRecursive(data, uniqueKeysSet) {
    if (Array.isArray(data)) {
        data.forEach(item => findAllUniqueKeysRecursive(item, uniqueKeysSet));
    } else if (typeof data === 'object' && data !== null) {
        for (const key in data) {
            if (data.hasOwnProperty(key)) {
                uniqueKeysSet.add(key); // Add the key itself
                findAllUniqueKeysRecursive(data[key], uniqueKeysSet); // Recurse into value
            }
        }
    }
    // Ignore primitives
}

function populateExtractKeysSelector() {
    extractKeysSelect.innerHTML = ''; // Clear previous options
    extractKeysSelect.disabled = true;
    addExtractBtn.disabled = true;

    if (!currentData) {
         extractKeysSelect.innerHTML = '<option value="">--Load JSON first--</option>';
        return;
    }

    const uniqueKeysSet = new Set();
    findAllUniqueKeysRecursive(currentData, uniqueKeysSet);

    const sortedKeys = Array.from(uniqueKeysSet).sort();

    if (sortedKeys.length > 0) {
        sortedKeys.forEach(key => {
            const option = document.createElement('option');
            option.value = key;
            option.textContent = key;
            extractKeysSelect.appendChild(option);
        });
        extractKeysSelect.disabled = false;
        addExtractBtn.disabled = false;
    } else {
        extractKeysSelect.innerHTML = '<option value="">--No keys found--</option>';
    }
}

function addExtractToQuery() {
    const selectedOptions = Array.from(extractKeysSelect.selectedOptions);
    if (selectedOptions.length === 0) {
        updateStatus('Please select at least one key to extract.', true);
        return;
    }

    const keysToExtract = selectedOptions.map(option => `'${option.value.replace(/'/g, "\\'")}'`); // Get value and quote it safely
    const extractPart = `extract(${keysToExtract.join(', ')})`;

    const currentQuery = queryInput.value.trim();
    if (currentQuery === '' || currentQuery.endsWith('|')) {
        queryInput.value = currentQuery + (currentQuery ? ' ' : '') + extractPart;
    } else {
        queryInput.value = currentQuery + ' | ' + extractPart;
    }
    updateStatus('Extract operation added to query input.', false);
}
